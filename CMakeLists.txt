cmake_minimum_required(VERSION 3.25)
project(IndustrialSim VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
function(sim_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
add_library(sim_core STATIC src/core/simulation.cpp)
target_include_directories(sim_core PUBLIC include)
sim_warnings(sim_core)
add_library(sim_runtime STATIC src/runtime/runner.cpp src/runtime/asset_cache.cpp)
target_include_directories(sim_runtime PUBLIC include)
target_link_libraries(sim_runtime PUBLIC sim_core Threads::Threads)
sim_warnings(sim_runtime)
add_executable(sim_headless apps/headless/main.cpp)
target_link_libraries(sim_headless PRIVATE sim_runtime)
sim_warnings(sim_headless)
if(BUILD_TESTING)
  set(SIM_GTEST_SOURCE_DIR "" CACHE PATH "Existing GoogleTest source directory")
  if(SIM_GTEST_SOURCE_DIR)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    add_subdirectory("${SIM_GTEST_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
  else()
    find_package(GTest CONFIG REQUIRED)
  endif()
  add_executable(sim_tests tests/simulation_test.cpp tests/runtime_test.cpp tests/cache_test.cpp)
  target_link_libraries(sim_tests PRIVATE sim_runtime GTest::gtest_main)
  sim_warnings(sim_tests)
  include(GoogleTest)
  gtest_discover_tests(sim_tests DISCOVERY_TIMEOUT 30 PROPERTIES TIMEOUT 15)
  add_test(NAME headless_demo COMMAND sim_headless)
endif()
