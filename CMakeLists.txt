cmake_minimum_required(VERSION 3.25)
project(IndustrialSim VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(SIM_BUILD_PROJECT "Build validated project documents (Windows)" ON)
find_package(Threads REQUIRED)
function(sim_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
add_library(sim_core STATIC src/core/simulation.cpp)
target_include_directories(sim_core PUBLIC include)
sim_warnings(sim_core)
add_library(sim_runtime STATIC src/runtime/runner.cpp src/runtime/asset_cache.cpp)
target_include_directories(sim_runtime PUBLIC include)
target_link_libraries(sim_runtime PUBLIC sim_core Threads::Threads)
sim_warnings(sim_runtime)
add_executable(sim_headless apps/headless/main.cpp)
target_link_libraries(sim_headless PRIVATE sim_runtime)
sim_warnings(sim_headless)

if(SIM_BUILD_PROJECT)
  if(NOT WIN32)
    message(FATAL_ERROR "Project crypto currently targets Windows; use -DSIM_BUILD_PROJECT=OFF for the portable core")
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The project document target requires Windows x64")
  endif()
  set(SIM_JSON_SOURCE_DIR "" CACHE PATH "Existing nlohmann/json source directory")
  set(SIM_JSON_SCHEMA_SOURCE_DIR "" CACHE PATH "Existing JSON schema validator source directory")
  if(SIM_JSON_SOURCE_DIR)
    set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
    set(JSON_Install OFF CACHE BOOL "" FORCE)
    add_subdirectory("${SIM_JSON_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/json" EXCLUDE_FROM_ALL)
  else()
    find_package(nlohmann_json CONFIG REQUIRED)
  endif()
  if(SIM_JSON_SCHEMA_SOURCE_DIR)
    set(JSON_VALIDATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(JSON_VALIDATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(JSON_VALIDATOR_INSTALL OFF CACHE BOOL "" FORCE)
    set(JSON_VALIDATOR_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    # Never allow the validator to fetch its own JSON dependency.
    set(FETCHCONTENT_FULLY_DISCONNECTED ON)
    add_subdirectory("${SIM_JSON_SCHEMA_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/json-schema" EXCLUDE_FROM_ALL)
  else()
    find_package(nlohmann_json_schema_validator CONFIG REQUIRED)
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/schemas/project-1.0.schema.json")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/schemas/project-1.0.schema.json" SIM_PROJECT_SCHEMA)
  configure_file(src/project/schema.hpp.in generated/project_schema.hpp @ONLY)
  add_library(sim_project_document STATIC src/project/document.cpp src/project/crypto_windows.cpp)
  target_include_directories(sim_project_document PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(sim_project_document PUBLIC sim_core PRIVATE nlohmann_json::nlohmann_json nlohmann_json_schema_validator::validator bcrypt)
  sim_warnings(sim_project_document)
  add_executable(sim_project apps/project/main.cpp)
  target_link_libraries(sim_project PRIVATE sim_project_document sim_runtime)
  if(MINGW)
    target_link_options(sim_project PRIVATE -municode)
  endif()
  sim_warnings(sim_project)
endif()

  endif()
